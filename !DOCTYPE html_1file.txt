    <script>
        /* ================= STATE GAME ================= */
        let stats = {
            hunger: 80,
            cleanliness: 90,
            energy: 75,
            happiness: 85,
            level: 1,
            xp: 0,
            coins: 100
        };

        let isSleeping = false;
        let isBathing = false;
        let isMuted = false;
        let audioCtx = null;

        // Inventory / Ownership Status
        let ownedAccessories = [];
        let currentAccessory = null;
        let currentBg = 'bg-pink';

        /* ================= WEB AUDIO API SYNTHESIZER ================= */
        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
        }

        function playSound(type) {
            if (isMuted) return;
            initAudio();
            if (!audioCtx) return;

            const now = audioCtx.currentTime;

            if (type === 'meow') {
                // Pitch Envelope Meow
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(400, now);
                osc.frequency.exponentialRampToValueAtTime(800, now + 0.15);
                osc.frequency.exponentialRampToValueAtTime(600, now + 0.35);

                gain.gain.setValueAtTime(0.3, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.35);

                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start(now);
                osc.stop(now + 0.35);

            } else if (type === 'eat') {
                // Sound efek Nyam-nyam
                for (let i = 0; i < 3; i++) {
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(250 + i * 50, now + i * 0.1);
                    gain.gain.setValueAtTime(0.2, now + i * 0.1);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + i * 0.1 + 0.08);

                    osc.connect(gain);
                    gain.connect(audioCtx.destination);
                    osc.start(now + i * 0.1);
                    osc.stop(now + i * 0.1 + 0.08);
                }

            } else if (type === 'coin') {
                // Sound efek dapat Koin
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(987.77, now); // B5
                osc.frequency.setValueAtTime(1318.51, now + 0.08); // E6

                gain.gain.setValueAtTime(0.2, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);

                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start(now);
                osc.stop(now + 0.25);

            } else if (type === 'soap') {
                // Sound efek Gelembung Sabun
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(300 + Math.random() * 200, now);
                osc.frequency.linearRampToValueAtTime(600, now + 0.05);

                gain.gain.setValueAtTime(0.15, now);
                gain.gain.linearRampToValueAtTime(0.01, now + 0.05);

                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start(now);
                osc.stop(now + 0.05);
            }
        }

        function toggleAudio() {
            isMuted = !isMuted;
            const icon = document.getElementById('sound-icon');
            if (isMuted) {
                icon.className = 'fa-solid fa-volume-xmark text-gray-400';
            } else {
                icon.className = 'fa-solid fa-volume-high text-pink-500';
                playSound('meow');
            }
        }

        /* ================= GAME LOOP & STATUS TICK ================= */
        setInterval(() => {
            if (!isSleeping) {
                stats.hunger = Math.max(0, stats.hunger - 1);
                stats.cleanliness = Math.max(0, stats.cleanliness - 0.8);
                stats.energy = Math.max(0, stats.energy - 0.5);
                stats.happiness = Math.max(0, stats.happiness - 0.7);
            } else {
                // Saat Tidur: Energi & Lapar berkurang sedikit
                stats.energy = Math.min(100, stats.energy + 4);
                stats.hunger = Math.max(0, stats.hunger - 0.3);
                if (stats.energy >= 100) {
                    toggleSleep(); // Bangun otomatis saat energi penuh
                    showDialog("Hoaam... Energi sudah penuh! 😺");
                }
            }

            updateUI();
        }, 3000); // Setiap 3 detik status diupdate

        /* ================= UPDATE UI & EKSPRESI KUCING ================= */
        function updateUI() {
            // Update Stat Bars
            document.getElementById('hunger-val').innerText = `${Math.round(stats.hunger)}%`;
            document.getElementById('hunger-bar').style.width = `${stats.hunger}%`;

            document.getElementById('clean-val').innerText = `${Math.round(stats.cleanliness)}%`;
            document.getElementById('clean-bar').style.width = `${stats.cleanliness}%`;

            document.getElementById('energy-val').innerText = `${Math.round(stats.energy)}%`;
            document.getElementById('energy-bar').style.width = `${stats.energy}%`;

            document.getElementById('happy-val').innerText = `${Math.round(stats.happiness)}%`;
            document.getElementById('happy-bar').style.width = `${stats.happiness}%`;

            // Update Level & Koin
            document.getElementById('level-num').innerText = stats.level;
            document.getElementById('level-txt').innerText = stats.level;
            document.getElementById('coin-txt').innerText = stats.coins;
            document.getElementById('xp-bar').style.width = `${stats.xp}%`;

            // Display Kotoran jika Kebersihan Rendah
            const dirtOverlay = document.getElementById('dirt-overlay');
            if (stats.cleanliness < 40) {
                dirtOverlay.classList.remove('hidden');
            } else {
                dirtOverlay.classList.add('hidden');
            }

            // Update Ekspresi Muka
            updateFaceExpression();
        }

        function updateFaceExpression() {
            // Sembunyikan Semua Mata & Mulut
            document.getElementById('eyes-normal').classList.add('hidden');
            document.getElementById('eyes-sleep').classList.add('hidden');
            document.getElementById('eyes-sad').classList.add('hidden');
            document.getElementById('eyes-happy').classList.add('hidden');

            document.getElementById('mouth-normal').classList.add('hidden');
            document.getElementById('mouth-happy').classList.add('hidden');
            document.getElementById('mouth-sad').classList.add('hidden');

            if (isSleeping) {
                document.getElementById('eyes-sleep').classList.remove('hidden');
                document.getElementById('mouth-normal').classList.remove('hidden');
            } else if (stats.hunger < 30 || stats.cleanliness < 30 || stats.happiness < 30) {
                // Ekspresi Sedih / Lapar / Kotor
                document.getElementById('eyes-sad').classList.remove('hidden');
                document.getElementById('mouth-sad').classList.remove('hidden');
            } else if (stats.happiness > 75) {
                // Ekspresi Sangat Senang
                document.getElementById('eyes-happy').classList.remove('hidden');
                document.getElementById('mouth-happy').classList.remove('hidden');
            } else {
                // Ekspresi Normal
                document.getElementById('eyes-normal').classList.remove('hidden');
                document.getElementById('mouth-normal').classList.remove('hidden');
            }
        }

        function addXP(amount) {
            stats.xp += amount;
            if (stats.xp >= 100) {
                stats.xp = 0;
                stats.level += 1;
                stats.coins += 50; // Bonus Level Up
                playSound('coin');
                createParticles('⭐');
                showDialog(`Hore! Naik ke Level ${stats.level}! +50 Koin 🎉`);
            }
        }

        function showDialog(text) {
            const dialog = document.getElementById('dialog-text');
            dialog.innerText = text;
        }

        /* ================= INTERAKSI KUCING ================= */
        function clickCat(e) {
            if (isSleeping) return;
            playSound('meow');
            addXP(2);
            stats.happiness = Math.min(100, stats.happiness + 2);
            createParticles('❤️', e.clientX, e.clientY);

            const randomDialogs = [
                "Purrr... Sayang kamu! ❤️",
                "Meow! Usap lagi dong~",
                "Aku kucing paling bahagia! 🐱",
                "Rawr! Aku macan kecil! 🐾"
            ];
            showDialog(randomDialogs[Math.floor(Math.random() * randomDialogs.length)]);
            updateUI();
        }

        /* ================= EFEK PARTIKEL ================= */
        function createParticles(symbol, x, y) {
            const room = document.getElementById('room-bg');
            const particle = document.createElement('div');
            particle.className = 'particle text-2xl';
            particle.innerText = symbol;

            if (x && y) {
                const rect = room.getBoundingClientRect();
                particle.style.left = `${x - rect.left}px`;
                particle.style.top = `${y - rect.top}px`;
            } else {
                particle.style.left = `${Math.random() * 80 + 10}%`;
                particle.style.top = `60%`;
            }

            room.appendChild(particle);
            setTimeout(() => particle.remove(), 1500);
        }

        /* ================= AKSI: MAKAN (FEED) ================= */
        function feedCat(foodType) {
            if (isSleeping) {
                showDialog("Kucing sedang tidur Zzz...");
                return;
            }

            if (foodType === 'milk') {
                stats.hunger = Math.min(100, stats.hunger + 15);
                playSound('eat');
                createParticles('🥛');
                showDialog("Slurp slurp... Susu segar lezat! 🥛");
            } else if (foodType === 'fish') {
                if (stats.coins < 10) {
                    showDialog("Koin tidak cukup untuk beli ikan! 🪙");
                    return;
                }
                stats.coins -= 10;
                stats.hunger = Math.min(100, stats.hunger + 35);
                playSound('eat');
                createParticles('🐟');
                showDialog("Nyam nyam! Ikan favoritku! 🐟");
            } else if (foodType === 'snack') {
                if (stats.coins < 15) {
                    showDialog("Koin tidak cukup untuk beli biskuit! 🪙");
                    return;
                }
                stats.coins -= 15;
                stats.hunger = Math.min(100, stats.hunger + 20);
                stats.happiness = Math.min(100, stats.happiness + 10);
                playSound('eat');
                createParticles('🍪');
                showDialog("Biskuit renyah! Enak sekali! 😋");
            }

            addXP(10);
            closeSubMenu();
            updateUI();
        }

        function startBath() {
            if (isSleeping) {
                showDialog("Kucing sedang tidur Zzz...");

        /* ================= AKSI: TIDUR (SLEEP) ================= */
        function toggleSleep() {
            isSleeping = !isSleeping;
            const container = document.getElementById('game-container');
            const sleepAnim = document.getElementById('sleep-animation');

            if (isSleeping) {
                container.classList.add('night-mode');
                sleepAnim.classList.remove('hidden');
                showDialog("Selamat tidur kucing manis... Zzz 🌙");

                container.classList.remove('night-mode');
                sleepAnim.classList.add('hidden');
                showDialog("Selamat pagi! Waktunya main lagi! ☀️");
            }
            updateUI();
        }

        /* ================= MINI GAME TANGKAP BENANG ================= */
        let gameScore = 0;
        let gameTimeLeft = 15;
        let gameInterval = null;
        let spawnInterval = null;

        function startMiniGame() {
            if (isSleeping) {
                showDialog("Kucing sedang tidur Zzz...");
                return;
            }

            document.getElementById('minigame-overlay').classList.remove('hidden');
            gameScore = 0;
            gameTimeLeft = 15;
            document.getElementById('game-score').innerText = gameScore;
            document.getElementById('game-timer').innerText = gameTimeLeft;

            const gameArea = document.getElementById('game-area');
            gameArea.innerHTML = '';

            gameInterval = setInterval(() => {
                gameTimeLeft--;
                document.getElementById('game-timer').innerText = gameTimeLeft;
                if (gameTimeLeft <= 0) {
                    endMiniGame();
                }
            }, 1000);

            spawnInterval = setInterval(spawnYarnBall, 700);
        }

        function spawnYarnBall() {
            const gameArea = document.getElementById('game-area');
            const ball = document.createElement('div');
            ball.className = 'absolute text-4xl cursor-pointer active:scale-125 transition-transform';
            ball.innerText = '🧶';

            const maxX = gameArea.clientWidth - 50;
            const maxY = gameArea.clientHeight - 50;
            ball.style.left = `${Math.random() * maxX}px`;
            ball.style.top = `${Math.random() * maxY}px`;

            ball.onclick = () => {
                gameScore += 5;
                stats.coins += 2;
                document.getElementById('game-score').innerText = gameScore;
                playSound('coin');
                ball.remove();
            };

            gameArea.appendChild(ball);

            setTimeout(() => {
                if (ball.parentNode) ball.remove();
            }, 1200);
        }

        function endMiniGame() {
            clearInterval(gameInterval);
            clearInterval(spawnInterval);
            document.getElementById('minigame-overlay').classList.add('hidden');

            stats.happiness = Math.min(100, stats.happiness + 30);
            addXP(20);
            showDialog(`Hore! Dapat ${gameScore} Skor & Bonus Koin! 🧶`);
            updateUI();
        }

        /* ================= SUB MENU OVERLAY (SHOP / FOOD) ================= */
        function openSubMenu(type) {
            const overlay = document.getElementById('overlay-menu');
            const title = document.getElementById('overlay-title');
            const feedContent = document.getElementById('content-feed');
            const shopContent = document.getElementById('content-shop');

            overlay.classList.remove('hidden');
            feedContent.classList.add('hidden');
            shopContent.classList.add('hidden');

            } else if (type === 'shop') {
                title.innerText = 'Toko Aksesori & Dekor 🛍️';
                shopContent.classList.remove('hidden');
            }
        }

        function closeSubMenu() {
            document.getElementById('overlay-menu').classList.add('hidden');
        }

        /* ================= SHOP SYSTEM (AKSESORI & BG) ================= */
        function buyAccessory(item, price) {
            if (stats.coins < price) {
                showDialog("Koin tidak cukup! 🪙");
                return;
            }

            const hatContainer = document.getElementById('accessory-hat');
            const glasses = document.getElementById('accessory-glasses');

            if (item === 'hat_party') {
                stats.coins -= price;
                hatContainer.innerText = '🥳';
                glasses.classList.add('hidden');
            } else if (item === 'hat_crown') {
                stats.coins -= price;
                hatContainer.innerText = '👑';
                glasses.classList.add('hidden');
            } else if (item === 'glasses') {
                stats.coins -= price;
                hatContainer.innerText = '';
                glasses.classList.remove('hidden');
            }

            playSound('coin');
            showDialog("Aksesori baru dipakai! Makin imut! ✨");
            closeSubMenu();
            updateUI();
        }

        function buyBackground(bgType, price) {
            if (stats.coins < price) {
                showDialog("Koin tidak cukup! 🪙");
                return;
            }

            stats.coins -= price;
            const roomBg = document.getElementById('room-bg');

            if (bgType === 'bg-pink') {
                roomBg.className = "relative flex-1 flex flex-col items-center justify-center p-4 transition-all duration-500 bg-cover bg-center";
            } else if (bgType === 'bg-garden') {
                roomBg.className = "relative flex-1 flex flex-col items-center justify-center p-4 transition-all duration-500 bg-emerald-200";
            }

            playSound('coin');
            showDialog("Latar belakang ruangan berhasil diubah! 🏡");
            closeSubMenu();
            updateUI();
        }
